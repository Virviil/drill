package main

func isPalindrome(s string) bool {
	ok := func(c byte) bool { return c >= 'a' && c <= 'z' || c >= 'A' && c <= 'Z' || c >= '0' && c <= '9' }
	low := func(c byte) byte {
		if c >= 'A' && c <= 'Z' {
			return c + 32
		}
		return c
	}
	for i, j := 0, len(s)-1; i < j; {
		switch {
		case !ok(s[i]):
			i++
		case !ok(s[j]):
			j--
		case low(s[i]) != low(s[j]):
			return false
		default:
			i, j = i+1, j-1
		}
	}
	return true
}
